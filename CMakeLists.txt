cmake_minimum_required(VERSION 3.24)
project(AVQualityMonitor VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
add_library(avqm_core src/detection/core.cpp)
target_include_directories(avqm_core PUBLIC src)
include(CTest)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE avqm_core)
add_test(NAME detection_core COMMAND core_tests)
option(AVQM_BUILD_APP "Build Windows media monitor" ON)
if(AVQM_BUILD_APP)
  set(FFMPEG_ROOT "${CMAKE_SOURCE_DIR}/third_party/ffmpeg")
  foreach(lib avformat avcodec avutil swscale swresample)
    find_library(FF_${lib} NAMES ${lib} PATHS "${FFMPEG_ROOT}/lib" "${FFMPEG_ROOT}/msvc" NO_DEFAULT_PATH REQUIRED)
    list(APPEND FF_LIBRARIES "${FF_${lib}}")
  endforeach()
  add_library(sqlite3 STATIC third_party/sqlite/sqlite3.c)
  target_include_directories(sqlite3 PUBLIC third_party/sqlite)
  add_executable(avqm src/app/main.cpp src/app/services.cpp src/app/logging.cpp src/config/settings.cpp src/media/chain.cpp src/media/rtsp_relay.cpp src/playback/output.cpp src/storage/store.cpp src/http/server.cpp)
  target_include_directories(avqm PRIVATE src third_party/include "${FFMPEG_ROOT}/include")
  target_link_libraries(avqm PRIVATE avqm_core sqlite3 ${FF_LIBRARIES} ws2_32 crypt32 bcrypt ole32 uuid avrt winmm gdi32 dwmapi d2d1 windowscodecs)
  set_target_properties(avqm PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/runtime")
  file(GLOB FF_DLLS "${FFMPEG_ROOT}/bin/*.dll")
  foreach(dll ${FF_DLLS})
    add_custom_command(TARGET avqm POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:avqm>")
  endforeach()
endif()
